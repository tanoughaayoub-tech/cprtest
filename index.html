// ── CPR Monitor Component (Updated with Student Defib Controls)
──────────
function CPRMonitor({session, onHome}){
const {roomCode, mode} = session; // mode will be 'instructor' or
'student'
const isInst = mode !== 'student';
// CPR state
const [cprActive,setCprActive] = useState(false);
const [comprRate,setComprRate] = useState(100);
const [etco2Quality,setEtco2Q] = useState(0.3);
const [airwayStep,setAirwayStep] = useState(0);
const [ventCycle,setVentCycle] = useState(0);
const [rhythm,setRhythm] = useState('VFib');
const [postShock,setPostShock] = useState(false);
// Defib state
const [joules,setJoules] = useState(200);
const [defibState,setDefibState] = useState('idle');
const [shockCount,setShockCount] = useState(0);
// Display values
const [dispETCO2,setDispETCO2] = useState(5);
const [dispHR,setDispHR] = useState(0);
const sRef =
useRef({cprActive,comprRate,etco2Quality,airwayStep,rhythm,defibState,jou
les,shockCount});
useEffect(()=>{sRef.current =
{cprActive,comprRate,etco2Quality,airwayStep,rhythm,defibState,joules,sho
ckCount};},[cprActive,comprRate,etco2Quality,airwayStep,rhythm,defibState
,joules,shockCount]);
// Update display ETCO2 based on quality
useEffect(()=>{
const val = airwayStep>=3 ? (5+etco2Quality*30) : 5;
setDispETCO2(Math.round(val));
setDispHR(rhythm==='Normal'?75:rhythm==='VFib'?0:0);
},[etco2Quality,airwayStep,rhythm]);
// Firebase sync - Instructor pushes state, student receives
useEffect(()=>{
if(!roomCode) return;
if(isInst){
// Instructor: Push all state to Firebase
const id = setInterval(()=>{
const s = sRef.current;
fw(roomCode, {
ecgRhythm: s.rhythm,
hr: s.rhythm==='Normal'?75:0,
spo2: s.airwayStep>=3?92:70,
perfusion: s.airwayStep>=3?0.5:0.1,
systolic: s.rhythm==='Normal'?90:0,
diastolic: s.rhythm==='Normal'?60:0,
etco2: 5 + s.etco2Quality*30,
rr: s.airwayStep>=3?10:0,
// CPR specific state
cprActive: s.cprActive,
comprRate: s.comprRate,
etco2Quality: s.etco2Quality,
airwayStep: s.airwayStep,
// Defib state
defibState: s.defibState,
joules: s.joules,
shockCount: s.shockCount
});
}, 500);
return () => clearInterval(id);
} else {
// Student: Listen to Firebase and update local state
const id = setInterval(async()=>{
const data = await fr(roomCode);
if(!data) return;
// Update all states from Firebase
setRhythm(data.ecgRhythm || 'VFib');
setCprActive(data.cprActive || false);
setComprRate(data.comprRate || 100);
setEtco2Q(data.etco2Quality || 0.3);
setAirwayStep(data.airwayStep || 0);
// Update defib state from instructor
setDefibState(data.defibState || 'idle');
setJoules(data.joules || 200);
setShockCount(data.shockCount || 0);
}, 300); // Faster sync for defib responsiveness
return () => clearInterval(id);
}
},[roomCode, isInst]);
// Generators
const rhythmHR =
{'Normal':75,'VTach':180,'VFib':300,'Asystole':0,'PEA':50};
const ecgGen = useCallback((t)=>
cprActive ? cprECGSample(t,comprRate) : ecgSample(t,rhythm,
rhythmHR[rhythm]||0),
[cprActive,comprRate,rhythm]);
const spo2Gen = useCallback((t)=>{
if(airwayStep===0) return 0.08 + (Math.random()-0.5)*0.01;
if(airwayStep<=2) return spo2Sample(t,75,0.12);
if(airwayStep===3) return spo2Sample(t,75,0.20);
return bvmSPO2Sample(t,8,0.55);
},[airwayStep]);
const ventStartRef = useRef(0);
useEffect(()=>{
if(ventCycle>0) ventStartRef.current = Date.now()/1000;
},[ventCycle]);
const etco2GenRef = useRef({ventCycle,etco2Quality,airwayStep});
useEffect(()=>{etco2GenRef.current={ventCycle,etco2Quality,airwayStep};},
[ventCycle,etco2Quality,airwayStep]);
const etco2Gen = useCallback((t)=>{
const {airwayStep:as, etco2Quality:eq} = etco2GenRef.current;
if(as<4) return 0.08 + (Math.random()-0.5)*0.008;
const elapsed = Date.now()/1000 - ventStartRef.current;
if(elapsed > 4.5) return 0.08 + (Math.random()-0.5)*0.008;
const maxH = Math.min(0.75,(4+eq*26)/60);
const breathPos = elapsed % 2;
if(breathPos<0.40){
const x=breathPos/0.40;
return 0.08+(maxH-0.08)*(1-Math.pow(1-x,1.8));
} else if(breathPos<0.90){
return maxH+(breathPos-0.40)*0.02;
} else if(breathPos<1.15){
const x=(breathPos-0.90)/0.25;
return (maxH+0.01)-(maxH+0.01-0.08)*Math.pow(x,0.3);
}
return 0.08;
},[]);
function handleCharge(){
if(defibState==='charging'||defibState==='ready') return;
setDefibState('charging');
audio.startCharge(joules,()=>{setDefibState('ready');});
}
function handleShock(){
if(defibState!=='ready') return;
audio.shock();
setDefibState('idle');
setShockCount(v=>v+1);
setPostShock(true);
setCprActive(false);
setTimeout(()=>setPostShock(false),500);
}
return(
<div className="monitor">
{/* Values column */}
{(()=>{
const spo2Display = airwayStep===0?'--
':airwayStep<=2?'52':airwayStep===3?'60':airwayStep>=4?'88':'--';
const spo2Alarm = airwayStep<4;
const etco2Display = airwayStep>=4 ?
Math.round(4+etco2Quality*26) : 4;
const rrDisplay = airwayStep>=4 ? 8 : 0;
return(
<div className="vals">
<div className="vcell">
<span className="vlbl ec">ECG</span>
<span className="vnum ec"
style={{fontSize:'1rem',marginTop:4}}>{rhythm==='Normal'?'ROSC':'—
'}</span>
<span className="vsub">{cprActive?'CPR':rhythm}</span>
</div>
<div className="vcell">
<span className="vlbl sc">SPO2</span>
<span className={`vnum sc
${spo2Alarm&&airwayStep>0?'alarm':''}`}
style={{fontSize:airwayStep===0?'1.5rem':'2rem'}}>
{spo2Display}
</span>
<span className="vsub">{airwayStep>0?'%':''}</span>
</div>
<div className="vcell">
<span className="vlbl bc">NIBP</span>
<span className="vnum bc"
style={{fontSize:'1.3rem'}}>{rhythm==='Normal'?'90/60':'—/—'}</span>
</div>
<div className="vcell">
<span className="vlbl xc">ETCO2</span>
<span className={`vnum xc
${etco2Display<10?'alarm':''}`}>{etco2Display}</span>
<span className="vsub">mmHg</span>
<span className="vlbl rc" style={{marginTop:4}}>RR</span>
<span className="vnum rc"
style={{fontSize:'1.5rem'}}>{rrDisplay}</span>
<span className="vlbl rc"
style={{marginTop:4,color:'#ff6666'}}>Shocks</span>
<span className="vnum rc"
style={{fontSize:'1.8rem',color:'#ff6666'}}>{shockCount}</span>
</div>
</div>
);
})()}
{/* Strips */}
<div className="strips">
<WaveCanvas gen={ecgGen} color={postShock?'#ffffff':'#00ff7f'}
label={cprActive?'CPR — ECG':'ECG II LEAD'} lineWidth={2.5} key="cpr-
ecg"/>
<WaveCanvas gen={spo2Gen} color="#00bfff" label="Pleth / BVM"
lineWidth={2.5} key="cpr-spo2"/>
<WaveCanvas gen={etco2Gen}color="#ffd700" label="CO2"
lineWidth={2.5} key="cpr-etco2"/>
</div>
{/* Controls - Different UI for Instructor vs Student */}
<div className="ctrls">
<div className="topbar">
<button className="btn-home"
onClick={()=>{audio.stopFlatline();audio.stopCharge();onHome();}}>🏠
Home</button>
<Clock/>
</div>
{roomCode&&(
<div className="rinfo">
<div className="rcode">{roomCode}</div>
<div className="rrole">{isInst ? 'CPR Instructor' : 'CPR
Student'}</div>
</div>
)}
{/* Defib Panel - VISIBLE TO BOTH INSTRUCTOR AND STUDENT */}
<div className="defib-panel">
<div className="defib-title">⚡ Defibrillator</div>
<div className="defib-joules">
{[200,300,360].map(j=>(
<button key={j} className={`defib-j-btn
${joules===j?'sel':''}`}
onClick={()=>{if(defibState==='idle')setJoules(j);}}
disabled={!isInst && defibState !== 'idle'}>
{j}J
</button>
))}
</div>
<button
className={`shock-btn ${defibState}`}
onClick={defibState==='idle'?handleCharge:defibState==='ready'?handleShoc
k:undefined}>
{defibState==='idle'?`⚡ Charge ${joules}J`:
defibState==='charging'?'Charging...':
'SHOCK — STAND CLEAR'}
</button>
{!isInst && defibState !== 'idle' && (
<div
style={{fontSize:'0.65rem',color:'#ffaa00',textAlign:'center',marginTop:'
4px'}}>
⚡ Instructor is controlling defibrillator
</div>
)}
</div>
<div className="cdiv"/>
{/* Instructor-only controls */}
{isInst ? (
<>
{/* CPR toggle */}
<div className="csec">
<div className="ctitle" style={{color:'#ff4444'}}>CPR</div>
<button className={`cbtn ${cprActive?'on':''}`}
onClick={()=>setCprActive(v=>!v)}
style={{background:cprActive?'#4a0000':'',borderColor:cprActive?'#ff4444'
:'',color:cprActive?'#ff6666':''}}>
{cprActive?'⏸ Stop CPR':'▶ Start CPR'}
</button>
{cprActive&&(
<div style={{display:'flex',gap:'3px',marginTop:'2px'}}>
{[80,100,120].map(r=>(
<button key={r} className={`cbtn
${comprRate===r?'on':''}`}
style={{flex:1,fontSize:'0.72rem'}}
onClick={()=>setComprRate(r)}>{r}/min</button>
))}
</div>
)}
</div>
<div className="cdiv"/>
{/* Airway */}
<div className="csec">
<div className="ctitle"
style={{color:'#00bfff'}}>Airway</div>
{[{label:'OPA
Inserted',step:1},{label:'Suction',step:2},{label:'BVM
Connected',step:3}].map(s=>(
<button key={s.step}
className={`airway-btn
${airwayStep>=s.step?'done':''}`}
onClick={()=>{
if(airwayStep>=s.step){
setAirwayStep(s.step-1);
} else if(s.step===1||airwayStep>=s.step-1){
setAirwayStep(s.step);
}
}}>
{airwayStep>=s.step?'✓ ':''}{s.label}
</button>
))}
{airwayStep>=3 && (
<button
onClick={()=>{
setVentCycle(v=>v+1);
setAirwayStep(4);
}}
style={{
width:'100%',padding:'6px 7px',
border:'2px solid #ffd700',
borderRadius:'5px',
background:'#1a1500',
color:'#ffd700',
fontFamily:"'Rajdhani',sans-serif",
fontSize:'0.82rem',fontWeight:700,
cursor:'pointer',letterSpacing:'1px',
textAlign:'center',marginTop:'4px',
boxShadow:'0 0 8px rgba(255,215,0,0.3)',
animation:airwayStep===3?'pulse 1.5s infinite':'none'
}}>
⚡ Ventilate — 2 Breaths
</button>
)}
<button className="airway-btn"
style={{color:'#ff6666',borderColor:'#4a1a1a',marginTop:'4px'}}
onClick={()=>setAirwayStep(0)}>Reset Airway</button>
</div>
<div className="cdiv"/>
{/* ETCO2 quality */}
<div className="csec">
<div className="ctitle xc">ETCO2 Quality</div>
<div className="etco2-quality">
{[['Poor',0],['Fair',0.5],['Good',1]].map(([l,v])=>(
<button key={l} className={`eq-btn
${etco2Quality===v?'on':''}`}
onClick={()=>setEtco2Q(v)}>{l}</button>
))}
</div>
</div>
<div className="cdiv"/>
{/* Post shock rhythm */}
<div className="csec">
<div className="ctitle ec">Post-Shock</div>
{[['VFib','VFib'],['VTach','VTach'],['Asystole','Asystole'],['PEA','PEA']
,['ROSC','Normal']].map(([l,v])=>(
<button key={v} className={`cbtn ${rhythm===v?'on':''}`}
onClick={()=>{setRhythm(v);setCprActive(false);}}>
{l}
</button>
))}
</div>
</>
) : (
// Student view - shows status but limited controls
<div style={{marginTop:'10px'}}>
<div className="csec">
<div className="ctitle" style={{color:'#ffaa00'}}>📊 CPR
Status</div>
<div
style={{background:'#0d2035',borderRadius:'8px',padding:'10px',marginTop:
'5px'}}>
<div style={{display:'flex',justifyContent:'space-
between',marginBottom:'6px'}}>
<span style={{color:'#ff8888'}}>CPR:</span>
<span
style={{color:cprActive?'#88ff88':'#ff8888',fontWeight:'bold'}}>{cprActiv
e?'ACTIVE':'INACTIVE'}</span>
</div>
<div style={{display:'flex',justifyContent:'space-
between',marginBottom:'6px'}}>
<span style={{color:'#88aaff'}}>Airway:</span>
<span style={{color:'#88ff88'}}>
{airwayStep===0?'None':
airwayStep===1?'OPA':
airwayStep===2?'OPA+Suction':
airwayStep===3?'BVM Ready':
'Ventilating'}
</span>
</div>
<div style={{display:'flex',justifyContent:'space-
between'}}>
<span style={{color:'#ffaa88'}}>ETCO2 Quality:</span>
<span style={{color:'#ffaa88'}}>
{etco2Quality===0?'Poor':etco2Quality===0.5?'Fair':'Good'}
</span>
</div>
</div>
</div>
<div className="cdiv"/>
<div
style={{textAlign:'center',color:'#6a8a6a',fontSize:'0.72rem',marginTop:1
0,padding:'0 5px'}}>
⚡ Use defibrillator above<br/>
Instructor controls airway and CPR
</div>
</div>
)}
</div>
</div>
);
}
